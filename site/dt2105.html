Dim sOpChatName, strTrig

Sub Main ()

      'on error resume next

         If curUser.bOperator Then
            curUser.SendData cstr("$Hello "& sOpChatName &"|")
         End If
         Exit Sub
      End If
     
      If Left(sCurData, 8) = "$GetINFO" Then
