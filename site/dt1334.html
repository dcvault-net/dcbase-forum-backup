Use this script:

Code: Select all

sub main()
  
  on error resume next
  Set IO = CreateObject("Scripting.FileSystemObject")
  Set file = IO.CreateTextFile("controls.txt", True)

  for each c in frmHub.Controls
    n = c.name
    t = ""
    t = c.caption
    i = -1
    i = c.index

    file.Write(n)
    if i>-1 then file.write("(" & i & ")")
    if t<>"" then file.write(" = """ & t & """")
    file.writeline
  next

  file.close
      
end sub
