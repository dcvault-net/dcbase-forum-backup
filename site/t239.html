	function( userp, line )
		-- this is a hack...
		if line == "$ADCGET list /// 0 -1 ZL1" then
			DC():PrintDebug( "Dropped user that tried to crash you via exploit!" )
			return true
		else
			return false
		end
	end
)
DC():PrintDebug(" ** exploitfix.lua loaded")
